
import { When, Then } from '@cucumber/cucumber';
import { pageFixture } from '../hooks/pageFixture';

When(
  'User clicks the {string} button',
  async function (buttonName: string) {
    await pageFixture.eventWorkflowPage.clickButton(buttonName);
  }
);

Then(
  'The {string} table contains two entries',
  async function (tableName: string) {
    await pageFixture.eventWorkflowPage.verifyTableRowCount(tableName, 2);
  }
);

Then(
  'Event #2 is the first row with status {string}',
  async function (status: string) {
    await pageFixture.eventWorkflowPage.verifyEventRowStatus(0, 'Event #2', status);
  }
);

Then(
  'Event #1 is the second row with status {string} or {string}',
  async function (status1: string, status2: string) {
    await pageFixture.eventWorkflowPage.verifyEventRowStatusOneOf(
      1,
      'Event #1',
      [status1, status2]
    );
  }
);
