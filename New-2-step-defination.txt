async function verifyFieldValue(
  fieldName: string,
  expectedValue: string
): Promise<void> {
  const page = pageFixture.page;

  const fieldLabel = page.getByText(fieldName, { exact: true });

  await expect(fieldLabel).toBeVisible();

  // TODO: Replace with the actual field container or
  // page-object locator used by your application.
  await expect(
    fieldLabel.locator('..')
  ).toContainText(expectedValue);
}

Then(
  'The Event History table contains an entry for {string}',
  async function (eventName: string) {
    const page = pageFixture.page;

    const eventRow = page
      .getByRole('row')
      .filter({ hasText: eventName });

    await expect(eventRow.first()).toBeVisible();
  }
);
